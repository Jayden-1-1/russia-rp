  const setEnvVar = async (idx, name, value) => {
    await page.evaluate((idx, n, v) => {
      const nameEl = document.querySelector("input[name=\"envVars." + idx + ".name\"]");
      const valEl = document.querySelector("textarea[name=\"envVars." + idx + ".value\"]");
      const nSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, "value").set;
      const vSetter = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, "value").set;
      if (nameEl && valEl) {
        nSetter.call(nameEl, n);
        nameEl.dispatchEvent(new Event("input", { bubbles: true }));
        vSetter.call(valEl, v);
        valEl.dispatchEvent(new Event("input", { bubbles: true }));
      }
    }, idx, name, value);
  };
  await setEnvVar(0, "DATABASE_URL", "file:/var/data/miss-rp.db");
  console.log("env1 done");
